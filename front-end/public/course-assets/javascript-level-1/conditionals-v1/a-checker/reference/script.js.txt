// We’re going to make a program which determines if a word inputted by the user contains the letter ‘a’ or not. First, get a word from the user. Next, determine whether or that that word contains an ‘a’ or not, and if it does, use printToScreen() to print it that it has an 'a'. If it does not have an 'a', use print out that it does not.

// Reference with explicit input handling

let word = prompt("Please enter a word:");
if (word === null) {
  printToScreen("Word entry canceled.");
} else if (word.trim() === "") {
  printToScreen("Enter a nonempty word.");
} else {
  let containsAnA = false;
  for (let i = 0; i < word.length; i++) {
    if (word[i] === "a") {
      containsAnA = true;
    }
  }
  if (containsAnA) {
    printToScreen(word + " has an 'a' in it!");
  } else {
    printToScreen(word + " does not have an 'a' in it!");
  }
}

// DO NOT ALTER CODE WRITTEN BELOW THIS LINE

function printToScreen(text) {
  d3.select("#output").append("p")
    .attr("class", "course-output").text(String(text));
}
