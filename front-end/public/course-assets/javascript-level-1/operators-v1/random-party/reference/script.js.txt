let day = Math.floor(Math.random() * 30) + 1;
let hour = Math.floor(Math.random() * 11) + 10;

  + hour + ":" + String(minute).padStart(2, "0") + ".");

function printToScreen(text) {
  d3.select("#output")
    .append("p")
    .attr("class", "course-output")
    .text(String(text));
}
