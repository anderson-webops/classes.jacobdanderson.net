// In this project, we’ll draw Captain America’s Shield with D3! Read the symbol documentation in the D3 docs to learn how to draw a star https://github.com/d3/d3-shape#symbols.

// Solution

let width = 750;
let height = 750;

// Create SVG element to append to
let svg = d3.select("body")
  .append("svg")
  .attr("width", width)
  .attr("height", height)
  .attr("viewBox", "0 0 750 750")
  .attr("role", "img")
  .attr("aria-label", "Captain America's Shield");

// Draw the outer red circle
svg.append("circle")
  .attr("fill", "red")
  .attr("cx", 200)
  .attr("cy", 200)
  .attr("r", 110);

// Draw the white ring
svg.append("circle")
  .attr("fill", "white")
  .attr("cx", 200)
  .attr("cy", 200)
  .attr("r", 90);

// Draw the inner red ring
svg.append("circle")
  .attr("fill", "red")
  .attr("cx", 200)
  .attr("cy", 200)
  .attr("r", 70);

// Draw the blue center
svg.append("circle")
  .attr("fill", "blue")
  .attr("cx", 200)
  .attr("cy", 200)
  .attr("r", 50);

let star = d3.symbol().type(d3.symbolStar).size(2000);

svg.append("path")
  .attr("d", star)
  .attr("fill", "white")
  .attr("transform", "translate(200,200)");
