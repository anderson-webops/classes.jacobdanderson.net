// You are trying to make up a sample password, but you want to do it completely randomly. Use ASCII number values and loops to create a random string of 10 lowercase letters. Then, create a string with 10 random numbers between 0-9. Then, iterate through both strings, and alternate between adding the letters of the two strings into a new string which will be the password. Use the printToScreen() function to print the new password.

// Solution

let word = "";
for (let i = 0; i < 10; i ++) {
  let num = (Math.floor(Math.random() * 26)) + 'a'.charCodeAt();
  let letter = String.fromCharCode(num);
  word += letter;
}

let numbers = "";

for (let i = 0; i < 10; i++){
  let ranNum = Math.floor(Math.random() * 10);
  numbers += ranNum;
}

let final = "";
for (let i = 0; i < 10; i++){
  final += word[i];
  final += numbers[i];
}

printToScreen(final);

// DO NOT ALTER CODE WRITTEN BELOW THIS LINE

function printToScreen(text) {
  d3.select("#output").append("p")
    .attr("class", "course-output").text(String(text));
}
