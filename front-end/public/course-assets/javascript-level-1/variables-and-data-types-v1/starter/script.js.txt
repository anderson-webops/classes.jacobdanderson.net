function printToScreen(text) {
  let description;
  if (text === null) {
    description = 'null is a primitive value; typeof null returns "object".';
  } else {
    description = String(text) + ' has type "' + typeof text + '".';
  }
  d3.select("#output")
    .append("p")
    .attr("class", "course-output")
    .text(description);
}
