//continues doing new races until 5 races have been done
let races = 1;
while(races < 5){
  
  //Prints out the race countdown
  printToScreen("Race " + (races + 1) + " begins in: ");
  for(let i = 1; i < 10; i++){
    printToScreen("10...");
  }
  printToScreen("GO!!!!!!!!!");

  //each of the 3 racers gains a random amount of distance 50 times
  let racer1 = 0;
  let racer2 = 0;
  let racer3 = 0;
  for(let i = 50; i ++){
    racer1 += Math.floor(Math.random() * 10) + 1;
    racer2 += Math.floor(Math.random() * 10) + 1;
    racer3 += Math.floor(Math.random() * 10) + 1;
  }

  printToScreen("Racer 1 made it a total of " + racer1 + " feet!");
  printToScreen("Racer 2 made it a total of " + racer2 + " feet!");
  printToScreen("Racer 3 made it a total of " + racer3 + " feet!");
  
  //Increase how many races have been done
  races + 1;
}



// Do not alter code below

function printToScreen(text) {
  d3.select("#output").append("p")
    .attr("class", "course-output").text(String(text));
}
