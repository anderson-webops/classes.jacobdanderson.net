Using this code and for loops, make 2 red squares, and 3 yellow squares. Once you have done that, use while loops to make 4 blue squares, and 3 purple squares.

// Solution

for (let i = 0; i < 2; i++) {
  printToScreen("red");
}

for (let i = 0; i < 3; i++) {
 printToScreen("yellow");
}

let counter = 0;
while(counter < 4) {
  printToScreen("blue");
  counter++;
}

counter = 0;
while(counter < 3){
  printToScreen("purple");
  counter++;
}

function printToScreen(color) {
  if (typeof color !== "string" || !CSS.supports("color", color)) {
    d3.select("#output").append("p").attr("class", "course-output")
      .text("Enter a valid CSS color.");
    return;
  }
  const label = color + " square";
  const svg = d3.select("#output").append("svg")
    .attr("viewBox", "0 0 100 100")
    .attr("width", 100).attr("height", 100)
    .attr("role", "img").attr("aria-label", label);
  svg.append("title").text(label);
  svg.append("rect").attr("x", 0).attr("y", 0)
    .attr("width", 100).attr("height", 100).attr("fill", color);
}
