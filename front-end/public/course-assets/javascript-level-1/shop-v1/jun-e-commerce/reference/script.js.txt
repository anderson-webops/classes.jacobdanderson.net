const carouselElement = document.querySelector("#featured-carousel");
if (carouselElement && window.M?.Carousel) {
  const preference = matchMedia("(prefers-reduced-motion: reduce)");
  const panels = [...carouselElement.querySelectorAll(".carousel-item")];
  function markCurrent(item) {
    const index = panels.indexOf(item);
    if (index < 0) return;
    carouselElement.dataset.activeFeature = String(index);
    panels.forEach(panel => { panel.inert = panel !== item; panel.setAttribute("aria-hidden", String(panel !== item)); });
    document.querySelector("#feature-status").textContent = "Photo " + (index + 1) + " of " + panels.length;
  }
  carouselElement.classList.add("carousel-ready");
  const carousel = M.Carousel.init(carouselElement, { fullWidth: true, indicators: true, duration: preference.matches ? 0 : 200, onCycleTo: markCurrent });
  markCurrent(panels[0]);
  carouselElement.querySelector(".indicators")?.setAttribute("aria-hidden", "true");
  document.querySelector(".feature-controls").hidden = false;
  document.querySelector("#previous-feature").addEventListener("click", () => carousel.prev());
  document.querySelector("#next-feature").addEventListener("click", () => carousel.next());
  preference.addEventListener("change", () => { carousel.options.duration = preference.matches ? 0 : 200; });
}
const productCards = [...document.querySelectorAll("[data-product]")];
if (productCards.length) {
  const products = new Map(productCards.map(card => [card.dataset.product, { name: card.querySelector("h3").textContent, cents: Number(card.dataset.priceCents) }]));
  const quantities = new Map();
  const list = document.querySelector("#cart-items");
  const summary = document.querySelector("#cart-summary");
  const currency = new Intl.NumberFormat("en-US", { style: "currency", currency: "USD" });
  const money = cents => currency.format(cents / 100);
  function renderCart() {
    let count = 0;
    let total = 0;
    list.replaceChildren();
    quantities.forEach((quantity, id) => {
      const product = products.get(id);
      count += quantity;
      total += product.cents * quantity;
      const row = document.createElement("li"); row.className = "cart-entry"; row.dataset.cartProduct = id;
      const description = document.createElement("div");
      const heading = document.createElement("h3"); heading.textContent = product.name;
      const amount = document.createElement("p"); amount.textContent = quantity + " × " + money(product.cents) + " = " + money(product.cents * quantity);
      description.append(heading, amount);
      const remove = document.createElement("button"); remove.type = "button"; remove.dataset.remove = id; remove.textContent = "Remove one"; remove.setAttribute("aria-label", "Remove one " + product.name);
      row.append(description, remove); list.append(row);
    });
    document.querySelector("#cart-count").textContent = String(count);
    document.querySelector("#cart-total").textContent = money(total);
    document.querySelector("#cart-empty").hidden = count !== 0;
    return { count, total };
  }
  productCards.forEach(card => {
    const button = card.querySelector("[data-add]");
    button.disabled = false;
    button.addEventListener("click", () => {
      const id = button.dataset.add;
      quantities.set(id, (quantities.get(id) ?? 0) + 1);
      const state = renderCart();
      document.querySelector("#cart-feedback").textContent = "Added " + products.get(id).name + ". Cart items: " + state.count + ". Total: " + money(state.total) + ".";
    });
  });
  list.addEventListener("click", event => {
    const button = event.target.closest("button[data-remove]");
    if (!button) return;
    const id = button.dataset.remove;
    const quantity = quantities.get(id);
    if (quantity > 1) quantities.set(id, quantity - 1); else quantities.delete(id);
    const state = renderCart();
    const replacement = [...list.querySelectorAll("button[data-remove]")].find(control => control.dataset.remove === id);
    (replacement ?? list.querySelector("button[data-remove]") ?? summary).focus();
    document.querySelector("#cart-feedback").textContent = "Removed one " + products.get(id).name + ". Cart items: " + state.count + ". Total: " + money(state.total) + ".";
  });
  renderCart();
  document.querySelector("#cart-panel").hidden = false;
}
