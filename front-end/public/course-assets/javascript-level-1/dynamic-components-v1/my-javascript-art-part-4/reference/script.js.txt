// Materialize 1.0 Sidenav supplies pointer gestures; add keyboard dismissal and a focus loop.
document.addEventListener("keydown", event => {
  const panel = document.querySelector("#slide-out");
  const instance = M.Sidenav.getInstance(panel);
  if (!instance?.isOpen) return;
  if (event.key === "Escape") { event.preventDefault(); instance.close(); return; }
  if (event.key !== "Tab") return;
  const controls = [...panel.querySelectorAll('a[href], button:not([disabled])')];
  const first = controls[0], last = controls[controls.length - 1];
  if (event.shiftKey && document.activeElement === first) { event.preventDefault(); last.focus(); }
  else if (!event.shiftKey && document.activeElement === last) { event.preventDefault(); first.focus(); }
});

M.Sidenav.init(document.querySelectorAll(".sidenav"), {

  onCloseEnd() { menuTrigger.setAttribute("aria-expanded", "false"); menuTrigger.focus(); }
});

const degreesToRadians = (degrees) => (
  degrees * Math.PI / 180
);

// Create a new svg to append to

  .append("svg")
  .attr("width", width2)

// Create the object that will generate our curves
const arcGenerator = d3.arc();

// Create the arc that will be the left Pac-Man–note the start and end angles, and their difference vs. the right Pac-Man below
let leftPacMan = arcGenerator({
  startAngle: degreesToRadians(120),
  endAngle: degreesToRadians(430),
  innerRadius: 0,
  outerRadius: 100
});

// Create the arc that will be the right Pac-Man
let rightPacMan = arcGenerator({
  startAngle: -degreesToRadians(120),
  endAngle: -degreesToRadians(430),
  innerRadius: 0,
  outerRadius: 100
});

  .attr("fill", "yellow")
  .attr("transform", "translate(150,110)")

  .attr("fill", "yellow")

function initializeScrollSpy() {
  const sections = document.querySelectorAll(".scrollspy");
  for (const section of sections) M.ScrollSpy.getInstance(section)?.destroy();
  const offset = document.querySelector("nav").getBoundingClientRect().height + 12;
  for (const section of sections) section.style.scrollMarginTop = offset + "px";
  M.ScrollSpy.init(sections, { scrollOffset: offset });
}

  if (!destination?.endsWith("-section")) return;
  event.preventDefault();
  event.stopPropagation();
  M.Dropdown.getInstance(document.querySelector("#artwork-trigger"))?.close();
  M.Sidenav.getInstance(document.querySelector("#slide-out"))?.close();
  location.hash = destination;
}, true);
