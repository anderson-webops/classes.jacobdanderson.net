const reducedMotion = matchMedia("(prefers-reduced-motion: reduce)").matches;

let canvasWidth = 1400;
let canvasHeight = 500;

let width = 500;
let height = 500;

let centerX = width / 2;
let centerY = height / 2;

let svg = d3
	.select("body")
	.append("svg")
	.attr("width", canvasWidth)
	.attr("height", canvasHeight)
	.attr("viewBox", `0 0 ${canvasWidth} ${canvasHeight}`)
	.attr("role", "img")
	.attr("aria-label", "Red car with two wheels, a windshield and a headlight");

let leftWheelX = centerX - width / 4;
let leftWheelY = centerY + height / 4;

let leftWheel = svg
	.append("circle")
	.attr("fill", "black")
	.attr("cx", leftWheelX)
	.attr("cy", leftWheelY)
	.attr("r", 50);

let leftWheelCap = svg
	.append("circle")
	.attr("fill", "grey")
	.attr("cx", leftWheelX)
	.attr("cy", leftWheelY)
	.attr("r", 40);

let rightWheelX = centerX + width / 4;
let rightWheelY = centerY + height / 4;

let rightWheel = svg
	.append("circle")
	.attr("fill", "black")
	.attr("cx", rightWheelX)
	.attr("cy", rightWheelY)
	.attr("r", 50);

let rightWheelCap = svg
	.append("circle")
	.attr("fill", "grey")
	.attr("cx", rightWheelX)
	.attr("cy", rightWheelY)
	.attr("r", 40);

let body = svg
	.append("rect")
	.attr("rx", 10)
	.attr("ry", 10)
	.attr("x", 50)
	.attr("y", 200)
	.attr("width", width - 50)
	.attr("height", 150)
	.attr("fill", "#ff2800");

let roof = svg
	.append("line")
	.attr("x1", width / 2 - 6)
	.attr("x2", (2 * width) / 3)
	.attr("y1", 105)
	.attr("y2", 105)
	.attr("stroke-width", 10)
	.attr("stroke", "black");

let windshield = svg
	.append("line")
	.attr("x1", (2 * width) / 3 - 6)
	.attr("x2", (2 * width) / 3 + 75)
	.attr("y1", 100)
	.attr("y2", 200)
	.attr("stroke-width", 10)
	.attr("stroke", "black");

let bodyTop = svg
	.append("rect")
	.attr("rx", 6)
	.attr("ry", 6)
	.attr("x", 50)
	.attr("y", 100)
	.attr("width", width / 2 - 50)
	.attr("height", 120)
	.attr("fill", "#ff2800");

let headlight = svg
	.append("rect")
	.attr("rx", 6)
	.attr("ry", 6)
	.attr("x", width - 25)
	.attr("y", 225)
	.attr("width", 20)
	.attr("height", 20)
	.attr("fill", "yellow");

svg
	.selectAll("*")
	.transition()
	.duration(reducedMotion ? 0 : 1500)
	.attr("transform", "translate(700)");
