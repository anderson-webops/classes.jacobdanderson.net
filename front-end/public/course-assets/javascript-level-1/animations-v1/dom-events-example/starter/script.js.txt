const reducedMotion = matchMedia("(prefers-reduced-motion: reduce)").matches;

let svg = d3
	.select("body")
	.append("svg")
	.attr("width", 750)
	.attr("height", 750)
	.attr("viewBox", "0 0 750 750");

let rect = svg
	.append("rect")
	.attr("fill", "red")
	.attr("x", 100)
	.attr("y", 100)
	.attr("width", 100)
	.attr("height", 100);

rect.attr("tabindex", 0).attr("role", "button").attr("aria-label", "Turn square pink");
function turnPink() {
  rect.interrupt().transition().duration(reducedMotion ? 0 : 250).attr("fill", "pink");
}
rect.on("click", turnPink);
rect.on("keydown", (event) => {
  if (event.key === "Enter" || event.key === " ") {
    event.preventDefault();
    turnPink();
  }
});
