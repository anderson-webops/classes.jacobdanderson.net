// Debug this fictional coffee-shop model using the prices and whole-pack requirements in README.md.


const gallonOfMilkCost = 2.49;
const bagOfBeansCost = 18.89;
const sackOfSugarCost = 9.99;
const costPerCup = 0.01;
const dailyMaintenanceCost = 30;
const dailyRentCost = 100;

const cupOfCoffeeCost = 1.99;

printToScreen(`Break-even: {reachProfitGoal(0)} cups of coffee.`);

printToScreen(`$200 profit goal: $reachProfitGoal(200) cups of coffee.`);

printToScreen(`$1000 profit goal: ${reachProfitGoal(1000)} cups of coffee.`);

// takes in the number of cups sold calculates the daily cost of selling that many cups of coffee
// Note: all ingredients go bad at the end of the day!
const dailyCost = (cupsSold) =>
  let gallonsOfMilkUsed = Math.ceiling(cups / 100);
  let bagsOfBeansUsed = Math.ceiling(cups / 150);
  let sacksOfSugarUsed = Math.ceiling(cups / 200);
  return (
    gallonsOfMilkUsed * gallonOfMilkCost +
    bagsOfBeansUsed * bagOfBeansCost +
    sacksOfSugarUsed * sackOfSugarCost +
    cupsSold * costPerCup +
    dailyMaintenanceCost +
    dailyRentCost
  );

// takes in the number of cups sold and returns a day's profit
const calculateProfit => {
  let cost = dailyCost(cupsSold);
  let revenue = cupOfCoffeeCost * cupsSold;
  printToScreen(revenue - cost);
};

// return the minimum number of cups to sell in order to reach a profit goal
const reachProfitGoal (goal) {
  let cups = 0;
  while (calculateProfit(cups) < goal) {
    cups++;
  }
  return cups;
};
