const canvas = document.getElementById("my-canvas");
const ctx = canvas.getContext("2d");
canvas.width = 550;
canvas.height = 500;
function randomInteger(min, max) {
    return Math.floor(Math.random() * (max - min + 1)) + min;
}
function randomRectangle() {
    const width = randomInteger(100, 299);
    const height = randomInteger(100, 299);
    return {
        x: randomInteger(50, canvas.width - 50 - width),
        y: randomInteger(50, canvas.height - 50 - height),
        width, height,
        color: "rgb(" + [randomInteger(0, 254), randomInteger(0, 254), randomInteger(0, 254)].join(",") + ")"
    };
}
function rectanglesTouch(a, b) {
    return a.x + a.width >= b.x && a.x <= b.x + b.width &&
        a.y + a.height >= b.y && a.y <= b.y + b.height;
}
const rect1 = randomRectangle();
const rect2 = randomRectangle();
for (const rect of [rect1, rect2]) {
    ctx.fillStyle = rect.color;
    ctx.fillRect(rect.x, rect.y, rect.width, rect.height);
}
const colliding = rectanglesTouch(rect1, rect2);
const message = colliding ? "The rectangles are colliding!" : "The rectangles are separated.";
ctx.fillStyle = "black";
ctx.font = "24px Arial";
ctx.textAlign = "center";
ctx.fillText(message, canvas.width / 2, 30);
canvas.setAttribute("aria-label", message);
