async function loadFox() {
	if (activeRequest || document.querySelector("#load").disabled) return;
	setBusy(true);
	setStatus("Loading fox.", "loading");
	try {
		const local = document.querySelector("#source").value === "saved";
		const fox = await loadFoxResponse(local);
		const image = await readyImage(fox.image);
		appendFox(fox, image);
		if (local) fixtureIndex++;
		setStatus(
			galleryCount + " fox " + (galleryCount === 1 ? "image." : "images."),
		);
	} catch (error) {
		showProblem(error, galleryCount > 0);
	} finally {
		setBusy(false);
	}
}
document.querySelector("#load").addEventListener("click", loadFox);
loadFox();
