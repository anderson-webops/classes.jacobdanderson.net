// Make an array called cards filled with strings which are 5 different names of colors. Set up a canvas, and then, use an iterator to iterate through cards to draw 5 cards in a row.

const canvas = document.getElementById("my-canvas");
const ctx = canvas.getContext("2d");
canvas.width = 1200;
canvas.height = 800;

const cards = ["green", "red", "blue", "yellow", "cyan"];

cards.forEach((color, index) => {
	let cardX = index * 200;
	let cardY = 150;
	ctx.beginPath();
	ctx.rect(cardX, cardY, 150, 100);
	ctx.fillStyle = color;
	ctx.fill();
	ctx.closePath();
});

document.getElementById("drawing-description").textContent =
	"Card order: " + cards.join(", ") + ".";
