// 2. Using arrayTwo, MODIFY THE ORIGINAL arrayTwo such that the array contains the ones digit from each original value. Use for-each.

const arrayTwo = [9, 1, 3, -10, 1092];
const problem2 = arr => {
	arr.forEach((val, index, _arr) => {
		_arr[index] = Math.abs(val) % 10;
	});
};

problem2(arrayTwo);
console.log(`problem two:  [9, 1, 3, -10, 1092] => [${arrayTwo}]`);

// 4. Use reduce to add all the even numbers of arrayThree together.

const arrayThree = [1, 9, 2, 3, 4, 10, 6, 29, -19, -28, 0, 1024];
const problem4 = arr => {
	return arr.reduce(
		(accumulator, val) => (val % 2 == 0 ? accumulator + val : accumulator),
		0
	);
};

console.log(`problem four:  [${arrayThree}] => [${problem4(arrayThree)}]`);

// 5. Use filter to get an make a function which returns an array of all the values in an arrayFour which are divisible.

const arrayFour = [1, 9, 2, 3, 4, 10, 6, 29, -19, -28, 0, 1024];
const problem5 = (arr, divisor) => {
	if (!Number.isInteger(divisor) || divisor === 0)
		throw new RangeError("Divisor must be a nonzero integer.");
	return arr.filter(val => val % divisor === 0);
};

console.log(`problem five:  [${arrayFour}] => [${problem5(arrayFour, 3)}]`);
