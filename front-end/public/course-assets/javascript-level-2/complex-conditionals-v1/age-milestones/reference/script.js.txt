// Five illustrative age conditions. These messages describe this exercise's thresholds.
const milestoneMessages = age => {
	if (!Number.isSafeInteger(age) || age < 0 || age > 150)
		throw new RangeError("Use a whole practice age from 0 to 150.");
	const isBaby =
		age <= 2 ? "Baby example (0–2): yes" : "Baby example (0–2): no";
	const isToddler =
		age > 2 && age < 5
			? "Toddler example (3–4): yes"
			: "Toddler example (3–4): no";
	const at16 =
		age >= 16 ? "Milestone at 16: reached" : "Milestone at 16: not yet";
	const at35 =
		age >= 35 ? "Milestone at 35: reached" : "Milestone at 35: not yet";
	const at65 =
		age >= 65 ? "Milestone at 65: reached" : "Milestone at 65: not yet";
	return ["Practice age: " + age, isBaby, isToddler, at16, at35, at65];
};
