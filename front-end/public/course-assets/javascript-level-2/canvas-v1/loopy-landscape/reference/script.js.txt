// Prompt: Draw a landscape on your canvas with trees, the sun, a river, etc. Make this project your own and construct a work of art.

const canvas = document.getElementById("my-canvas");

canvas.width = 800;
canvas.height = 800;

const ctx = canvas.getContext("2d");

ctx.fillStyle = "blue";
ctx.fillRect(0, 0, 800, 500);
ctx.fillStyle = "green";
ctx.fillRect(0, 500, 800, 300);

const drawTree = (x, y) => {
	ctx.beginPath();
	ctx.moveTo(x, y);
	ctx.lineTo(x + 50, y - 400);
	ctx.lineTo(x + 100, y);
	ctx.lineTo(x, y);
	ctx.closePath();
	ctx.fillStyle = "rgba(0,255,0)";
	ctx.fill();
	ctx.fillStyle = "brown";
	ctx.fillRect(x + 40, y - 50, 20, 250);
};

for (let i = 0; i < 3; i++) {
	drawTree(10 + 100 * i, i === 1 ? 550 : 600);
}

ctx.beginPath();
ctx.arc(600, 200, 100, 0, Math.PI * 2);
ctx.fillStyle = "yellow";
ctx.fill();
const drawingDescription =
	"Three trees beneath a yellow sun, with blue sky and green ground.";
document.getElementById("drawing-description").textContent = drawingDescription;
