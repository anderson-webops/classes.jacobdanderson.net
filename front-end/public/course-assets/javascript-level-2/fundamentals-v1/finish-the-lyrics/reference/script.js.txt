// Three original sayings are used for this version of the original assignment.
alert("Let's play a three-question word game!");
let numCorrect = 0;
let numLyrics = 3;

let lyric1 = prompt("A tiny bug can hide in a large _____.");
if (lyric1 !== null && lyric1.trim().toLowerCase() === "program") {
  numCorrect++;
  alert("Correct!");
} else {
  alert("The missing word was program.");
}

let lyric2 = prompt("A saved copy makes tomorrow's work _____.");
if (lyric2 !== null && lyric2.trim().toLowerCase() === "easier") {
  numCorrect++;
  alert("Correct!");
} else {
  alert("The missing word was easier.");
}

let lyric3 = prompt("A clear name makes a variable's purpose _____.");
if (lyric3 !== null && lyric3.trim().toLowerCase() === "clear") {
  numCorrect++;
  alert("Correct!");
} else {
  alert("The missing word was clear.");
}

printToScreen("You got " + numCorrect + " answers out of " + numLyrics + " correct!");

// Display helper: keep project logic above this line.
function printToScreen(text) {
  const colors = ["#0b2022", "#1235a8", "#4e175e", "#24551c"];
  const color = colors[Math.floor(Math.random() * colors.length)];
  const output = d3.select("body").append("p")
    .attr("id", "quiz-score")
    .attr("role", "status")
    .style("color", color)
    .style("transform", "translateX(-16px)")
    .style("opacity", 0)
    .text(text);
  const reducedMotion = window.matchMedia("(prefers-reduced-motion: reduce)").matches;
  output.transition().duration(reducedMotion ? 0 : 1000)
    .style("transform", "translateX(0px)")
    .style("opacity", 1);
}
